1.Получить список пользователей и количество их заказов.
SELECT u.id, u.email, COUNT(o.id) AS order_count
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id, u.email;

2.Вывести все товары с названием категории.
SELECT p.id, p.name AS product_name, p.price, c.name AS category_name
FROM products p
LEFT JOIN categories c ON p.category_id = c.id;

3.Найти пользователей, у которых есть хотя бы один заказ.
SELECT DISTINCT u.id, u.email, u.created_at
FROM users u
INNER JOIN orders o ON u.id = o.user_id;

4.Получить список заказов с email пользователя.
SELECT o.id, o.status, o.created_at, u.email

5.Вывести товары и количество отзывов по каждому товару.
SELECT p.id, p.name AS product_name, p.price, COUNT(r.id) AS review_count
FROM products p
LEFT JOIN reviews r ON p.id = r.product_id
GROUP BY p.id, p.name, p.price;

8.Найти заказы, в которых больше одной позиции.
SELECT o.id, o.user_id, o.status, o.created_at, COUNT(oi.id) AS items_count
FROM orders o
INNER JOIN order_items oi ON o.id = oi.order_id
GROUP BY o.id, o.user_id, o.status, o.created_at
HAVING COUNT(oi.id) > 1;

INNER JOIN countries cn ON ct.country_id = cn.id;

32.Найти пользователей, у которых количество заказов больше среднего.
SELECT u.id, u.email, COUNT(o.id) AS order_count
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
GROUP BY u.id, u.email
HAVING COUNT(o.id) > (SELECT AVG(order_count) FROM (SELECT COUNT(*) AS order_count FROM orders GROUP BY user_id) AS avg_orders);

33.Найти заказы с суммой больше средней суммы заказа.
SELECT o.id, o.user_id, SUM(oi.price * oi.quantity) AS order_total
FROM orders o
INNER JOIN order_items oi ON o.id = oi.order_id
GROUP BY o.id, o.user_id
HAVING SUM(oi.price * oi.quantity) > (SELECT AVG(order_total) FROM (SELECT SUM(price * quantity) AS order_total FROM order_items GROUP BY order_id) AS avg_order_totals);

34.Найти товары с рейтингом выше среднего.

GROUP BY p.id, p.name
HAVING AVG(r.rating) > (SELECT AVG(rating) FROM reviews);

45.Найти самый популярный товар в каждой категории.
SELECT c.name AS category_name, p.name AS product_name, MAX(pop.total_sold) AS max_sold
FROM categories c
INNER JOIN (
    SELECT p.category_id, p.id, p.name, SUM(oi.quantity) AS total_sold
    FROM products p
    INNER JOIN order_items oi ON p.id = oi.product_id
    GROUP BY p.category_id, p.id, p.name
) pop ON c.id = pop.category_id
GROUP BY c.id, c.name;

46.Найти пользователей, которые покупали товары минимум из двух категорий.
SELECT u.id, u.email, COUNT(DISTINCT p.category_id) AS categories_purchased
FROM users u
INNER JOIN orders o ON u.id = o.user_id
INNER JOIN order_items oi ON o.id = oi.order_id
INNER JOIN products p ON oi.product_id = p.id
GROUP BY u.id, u.email
HAVING COUNT(DISTINCT p.category_id) >= 2;

48.Найти пользователей, которые оставили отзывы на купленные товары.
SELECT DISTINCT u.id, u.email
FROM users u
INNER JOIN reviews r ON u.id = r.user_id
WHERE EXISTS (
    SELECT 1 FROM orders o
    INNER JOIN order_items oi ON o.id = oi.order_id
    WHERE o.user_id = u.id AND oi.product_id = r.product_id
);
